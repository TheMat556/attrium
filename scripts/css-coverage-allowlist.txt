# Suppressions for scripts/check-css-coverage.mjs (one selector per line).
#
# A selector lands in the DEAD report when it matches nothing on any crawled
# screen — but the crawler never hovers, opens menus/modals, drags, or types,
# so anything kept alive ONLY by such a transient state looks dead. If you
# verify a reported selector styles a real interactive state, pin it here
# instead of deleting the rule. Lines are matched verbatim (whitespace
# normalized) against the selector as written in the compiled chunk.
#
# Examples (commented out — uncomment after verifying):
# body.attrium-mod-screens .media-modal button.media-modal-close:hover

# Verified suppressions (check-css-coverage would otherwise flag these on
# every run, although each styles something real):
#
# - ::-moz-focus-inner only exists in Firefox; a Chromium crawl can never
#   match it, but the rule is load-bearing there.
body.attrium-mod-buttons #font-library-wp-admin-app .components-button.is-primary::-moz-focus-inner
body.attrium-mod-screens .connectors-page .components-button.is-primary::-moz-focus-inner
body.attrium-mod-screens .connectors-page .components-button.is-secondary::-moz-focus-inner
#
# - Update rows only render when updates are available; the determinism
#   mu-plugin freezes update transients, so the crawl never sees them.
body.attrium-mod-tables .widefat.plugins tr:has(+tr.plugin-update-tr) td
body.attrium-mod-tables .widefat.plugins tr:has(+tr.plugin-update-tr) th
